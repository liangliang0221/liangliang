硬性要求：
selenium不支持4.0.0以下版本，包括4.0.0
appium-python-client为2.11.0


1.点击Terminal位于当前目录下生成requirements.txt文件，执行如下命令生成依赖文件
pip freeze > requirements.txt
安装指令：pip install -r requirements.txt

2.配置文件详解
[pytest]
;命令行参数
;--html=./report/report.html    生成测试报告
;--reruns=3    用例失败后重跑次数
;--maxfail=3    出现几个用例失败报错，则终止测试
;-k 'xxx or xxx'  模糊匹配，运行测试用例名称中包含某个字符串的测试用例
addopts = -vs --html=./report/report.html --reruns=3 --maxfail=3

;注册自定义标记
markers =
    demo : marks tests as demo
    smoke: marks tests as smoke
    uat : marks tests as uat
    test : marks tests as test

;指定运行测试所需的最小pytest版本。
minversion = 5.0

;测试用例搜索范围
testpaths =./testcase
python_files = test_*  *_test  test*
python_classes = Test*   test*
python_functions = test_*  test*

;设置控制台输出样式：
console_output_style = progress

;存放日志到指定目录以及等级、展示样式、携带信息
log_file =./log/pytest_log.txt
log_file_level = info
log_file_format = %(asctime)s [%(levelname)s] | %(filename)s:%(lineno)s | %(message)s
log_file_date_format = %Y-%m-%d %H:%M:%S
